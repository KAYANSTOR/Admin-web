rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // دالة مساعدة للتحقق مما إذا كان المستخدم مديراً
    function isAdmin() {
      return request.auth != null && exists(/databases/$(database)/documents/admins/$(request.auth.uid));
    }

    // 1. الإعدادات العامة
    match /app_settings/global_config {
      allow read: if true;
      allow write: if isAdmin();
      
      match /notifications/{notificationId} {
        allow read: if request.auth != null;
        allow write: if isAdmin();
      }
    }

    // 2. إعدادات المستخدمين
    match /users/{userId} {
      allow read: if true;
      allow write: if isAdmin(); 
      
      match /notifications/{notificationId} {
        allow read: if isAdmin() || (request.auth != null && request.auth.uid == userId);
        allow update: if request.auth != null && request.auth.uid == userId && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['is_read']);
        allow write: if isAdmin();
      }
    }

    // 3. المبيعات الخاصة بأصحاب الشبكات
    match /networks/{userId} {
      allow read: if true;
      allow write: if request.auth != null && request.auth.uid == userId;
      
      match /sales/{saleId} {
        allow read: if true;
        allow write: if request.auth != null && request.auth.uid == userId;
      }
      
      match /_metadata/{document=**} {
        allow read: if true;
        allow write: if request.auth != null && request.auth.uid == userId;
      }
    }

    // 4. الاشتراكات والعمولات - تمت إضافتها لأن التطبيق يحاول قراءتها في لوحة التحكم
    match /subscriptions/{subscriptionId} {
      allow read: if true;
      allow write: if isAdmin();
    }
    
    match /commissions/{commissionId} {
      allow read: if true;
      allow write: if isAdmin();
    }
    
    // 5. المجموعات الفرعية للمبيعات (Collection Groups)
    match /{path=**}/sales/{saleId} {
      allow read: if true;
    }
  }
}
